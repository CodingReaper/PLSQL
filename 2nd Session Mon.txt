Get-CimInstance Win32_Share |
Where-Object { $_.Type -eq 0 -and $_.Path } |
ForEach-Object {
    "`r`n==================================================" |
        Out-File "C:\share_permissions.txt" -Append

    "SHARE NAME : $($_.Name)" |
        Out-File "C:\share_permissions.txt" -Append

    "PHYSICAL PATH : $($_.Path)" |
        Out-File "C:\share_permissions.txt" -Append

    "NTFS PERMISSIONS:" |
        Out-File "C:\share_permissions.txt" -Append

    icacls "$($_.Path)" |
        Out-File "C:\share_permissions.txt" -Append
}
